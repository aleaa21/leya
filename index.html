const GRID_SIZE = 13; // Koordinat yang dijamin 100% pas secara sistematis & tidak akan ada huruf bentrok const dataSoal = [ // Mendatar (H) { word: "STARS", r: 0, c: 0, dir: "H", num: 1 }, { word: "PYTHON", r: 2, c: 5, dir: "H", num: 3 }, { word: "WISDOM", r: 6, c: 0, dir: "H", num: 2 }, { word: "ANALYSIS", r: 10, c: 1, dir: "H", num: 5 }, { word: "EXCEL", r: 11, c: 3, dir: "H", num: 8 }, // Menurun (V) { word: "JUDGEMENT", r: 0, c: 2, dir: "V", num: 7 }, { word: "PORTOFOLIO", r: 0, c: 9, dir: "V", num: 4 }, { word: "ETHICS", r: 5, c: 9, dir: "V", num: 6 }, { word: "DATA", r: 6, c: 3, dir: "V", num: 9 }, { word: "SQL", r: 9, c: 7, dir: "V", num: 10 } ]; let grid = Array(GRID_SIZE).fill(null).map(() => Array(GRID_SIZE).fill(null)); dataSoal.forEach(s => { let row = s.r; let col = s.c; for (let i = 0; i < s.word.length; i++) { if (!grid[row][col]) { grid[row][col] = { letter: s.word[i], num: (i === 0 ? s.num : null) }; } else if (i === 0) { grid[row][col].num = s.num; } if (s.dir === "H") col++; else row++; } }); const tbody = document.getElementById("board-body"); tbody.innerHTML = ""; // Clear board sebelumnya for (let r = 0; r < GRID_SIZE; r++) { const tr = document.createElement("tr"); for (let c = 0; c < GRID_SIZE; c++) { const td = document.createElement("td"); const cellData = grid[r][c]; if (cellData) { td.id = `cell-${r}-${c}`; if (cellData.num) { const numSpan = document.createElement("span"); numSpan.className = "cell-num"; numSpan.innerText = cellData.num; td.appendChild(numSpan); } const input = document.createElement("input"); input.maxLength = 1; input.dataset.r = r; input.dataset.c = c; input.oninput = function() { this.value = this.value.toUpperCase(); }; td.appendChild(input); } else { td.className = "black"; } tr.appendChild(td); } tbody.appendChild(tr); }
